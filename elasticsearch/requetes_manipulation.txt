# ---------- 2. Créer l'index et définir sa structure (mapping) ----------
# keyword = valeur exacte (adresses, identifiants) ; double = nombre à virgule ;
# long = nombre entier ; date = date
PUT btc-transactions
{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 1
  },
  "mappings": {
    "properties": {
      "tx_hash":             { "type": "keyword" },
      "date":                { "type": "date" },
      "adresse_source":      { "type": "keyword" },
      "adresse_destination": { "type": "keyword" },
      "montant_btc":         { "type": "double" },
      "frais_btc":           { "type": "double" },
      "bloc":                { "type": "long" },
      "label_destination":   { "type": "keyword" }
    }
  }
}

# Vérifier la structure
GET btc-transactions/_mapping

# ---------- 5. Recherches ----------

# 5.1 Qui a envoyé de l'argent au relais ? (attendu : 5 résultats)
GET btc-transactions/_search
{
  "query": {
    "term": { "adresse_destination": "bc1qrelais99" }
  }
}

# 5.2 Gros montants, 1 BTC ou plus (attendu : 1 résultat, tx006)
GET btc-transactions/_search
{
  "query": {
    "range": { "montant_btc": { "gte": 1 } }
  }
}

# 5.3 Profil complet d'une adresse, par ordre chronologique
# (attendu : 6 résultats, 5 entrées entre 03:02 et 03:07 puis 1 sortie à 03:13)
GET btc-transactions/_search
{
  "query": {
    "bool": {
      "should": [
        { "term": { "adresse_source": "bc1qrelais99" } },
        { "term": { "adresse_destination": "bc1qrelais99" } }
      ]
    }
  },
  "sort": [ { "date": "asc" } ]
}


# ---------- 6. Agrégation : détecter le collecteur automatiquement ----------
# Regroupe par adresse de destination, avec le total reçu et le nombre de sources
# (attendu en tête : bc1qrelais99, 5 transactions, ~2.03 BTC, 5 sources)
GET btc-transactions/_search
{
  "size": 0,
  "aggs": {
    "par_destination": {
      "terms": { "field": "adresse_destination" },
      "aggs": {
        "total_recu": { "sum": { "field": "montant_btc" } },
        "nb_sources": { "cardinality": { "field": "adresse_source" } }
      }
    }
  }
}
